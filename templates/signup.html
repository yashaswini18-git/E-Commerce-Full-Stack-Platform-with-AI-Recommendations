# ========================================================= # SIGNUP # ========================================================= @app.route("/signup", methods=["GET", "POST"]) def signup(): if request.method == "POST": name = request.form.get("name") email = request.form.get("email") phone = request.form.get("phone") password = request.form.get("password") # Check if email already exists existing_user = User.query.filter_by( email=email ).first() if existing_user: flash( "An account with this email already exists." ) return redirect( url_for("signup") ) # Hash password before storing it hashed_password = generate_password_hash( password ) # Create new user new_user = User( name=name, email=email, password=hashed_password, phone=phone ) db.session.add( new_user ) db.session.commit() flash( "Account created successfully! Please login." ) return redirect( url_for("login") ) return render_template( "signup.html" )
